# CMakeLists_Tether.txt - Build the minimal TetherCli helper
#
# BUILD INSTRUCTIONS (from the sony-sdk root):
#
#   cd /Users/ssmacpro/dev/cam/native/sony-sdk
#   mkdir -p build_tether && cd build_tether
#   cmake -DCMAKE_BUILD_TYPE=Release -DSDK_ROOT=.. -f .. ../CMakeLists_Tether.txt 2>/dev/null || \
#     (ln -sf ../CMakeLists_Tether.txt CMakeLists.txt && cmake -DCMAKE_BUILD_TYPE=Release -DSDK_ROOT=.. .)
#   make -j$(sysctl -n hw.ncpu 2>/dev/null || nproc)
#
# The simplest reliable method:
#
#   cd /Users/ssmacpro/dev/cam/native/sony-sdk
#   mkdir -p build_tether && cd build_tether
#   ln -sf ../CMakeLists_Tether.txt CMakeLists.txt
#   cmake -DCMAKE_BUILD_TYPE=Release -DSDK_ROOT=$(cd .. && pwd) .
#   make -j8
#

cmake_minimum_required(VERSION 3.10)

if(APPLE)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "12.1" CACHE STRING "Minimum OS X deployment version")
endif()

project(TetherCli LANGUAGES CXX)

# SDK_ROOT must point to the sony-sdk directory.
# Auto-detect: if not set, assume parent of CMAKE_CURRENT_SOURCE_DIR or
# that we are being built from within build_tether/ inside the SDK root.
if(NOT DEFINED SDK_ROOT)
    # Check if CMAKE_CURRENT_SOURCE_DIR itself is the SDK root
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/app/CRSDK/CameraRemote_SDK.h")
        set(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}")
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../app/CRSDK/CameraRemote_SDK.h")
        get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
    else()
        message(FATAL_ERROR "Cannot locate SDK. Set -DSDK_ROOT=/path/to/sony-sdk")
    endif()
endif()

get_filename_component(SDK_ROOT "${SDK_ROOT}" ABSOLUTE)
message(STATUS "SDK_ROOT = ${SDK_ROOT}")

set(APP_DIR ${SDK_ROOT}/app)
set(CRSDK_HDR_DIR ${APP_DIR}/CRSDK)

### Source files ###
set(tether_srcs
    ${APP_DIR}/TetherCli.cpp
)

### CRSDK headers (for IDE indexing) ###
file(GLOB crsdk_hdrs ${CRSDK_HDR_DIR}/*.h)

### Define the executable ###
add_executable(TetherCli
    ${tether_srcs}
    ${crsdk_hdrs}
)

set_target_properties(TetherCli PROPERTIES
    CXX_STANDARD 17
    CXX_STANDARD_REQUIRED YES
    CXX_EXTENSIONS NO
)

if(APPLE)
    set_target_properties(TetherCli PROPERTIES
        BUILD_RPATH "@executable_path"
    )
endif()

if(UNIX AND NOT APPLE)
    set_target_properties(TetherCli PROPERTIES
        BUILD_RPATH "$ORIGIN"
        INSTALL_RPATH "$ORIGIN"
    )
    target_compile_options(TetherCli
        PUBLIC -fstack-protector-all
        PRIVATE -fsigned-char
    )
endif()

### Include directories ###
target_include_directories(TetherCli
    PRIVATE
        ${APP_DIR}
)

### Link the Camera Remote SDK core library ###
set(cr_ldir ${SDK_ROOT}/external/crsdk)
find_library(camera_remote Cr_Core HINTS ${cr_ldir})
target_link_libraries(TetherCli PRIVATE ${camera_remote})

### Linux: link filesystem if needed ###
if(UNIX AND NOT APPLE)
    if("${CMAKE_CXX_COMPILER_ID}" STREQUAL "GNU")
        if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 8)
            target_compile_definitions(TetherCli PRIVATE USE_EXPERIMENTAL_FS)
        endif()
        if(CMAKE_CXX_COMPILER_VERSION VERSION_LESS 9)
            target_link_libraries(TetherCli PRIVATE stdc++fs)
        endif()
    endif()
endif()

### Copy SDK runtime libraries next to the built binary ###
if(APPLE)
    set(ct_fw "Contents/Frameworks")
    set(adaptor_dir_name "CrAdapter")
    add_custom_command(TARGET TetherCli PRE_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory $<TARGET_FILE_DIR:TetherCli>/${ct_fw}/${adaptor_dir_name}
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${cr_ldir} $<TARGET_FILE_DIR:TetherCli>
        COMMAND ${CMAKE_COMMAND} -E copy_directory $<TARGET_FILE_DIR:TetherCli>/${adaptor_dir_name} $<TARGET_FILE_DIR:TetherCli>/${ct_fw}/${adaptor_dir_name}
        COMMAND ${CMAKE_COMMAND} -E rm -rf $<TARGET_FILE_DIR:TetherCli>/${adaptor_dir_name}
    )
endif()

if(UNIX AND NOT APPLE)
    add_custom_command(TARGET TetherCli PRE_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${cr_ldir} $<TARGET_FILE_DIR:TetherCli>
    )
endif()

if(WIN32)
    add_custom_command(TARGET TetherCli PRE_BUILD
        COMMAND ${CMAKE_COMMAND} -E copy_directory ${cr_ldir} $<TARGET_FILE_DIR:TetherCli>
    )
    target_compile_definitions(TetherCli PRIVATE UNICODE _UNICODE)
endif()

### Install ###
install(TARGETS TetherCli DESTINATION .)
install(DIRECTORY ${cr_ldir}/ DESTINATION .)
